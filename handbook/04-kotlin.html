3. Kotlin：可讲明白的 Warmup + Join 完整示例

下面示例强调架构和并发语义。真实 ACS/WebRTC API 用接口隔离，便于单测。核心能力：single-flight、TTL、网络 epoch、超时、失败熔断、取消安全、指标。

import kotlinx.coroutines.*
import kotlinx.coroutines.sync.Mutex
import kotlinx.coroutines.sync.withLock
import java.time.Clock
import java.time.Duration
import java.time.Instant
import java.util.UUID
import java.util.concurrent.atomic.AtomicLong

data class VoiceKey(val userId: String, val tenantId: String, val configVersion: String)
data class Credential(val token: String, val expiresAt: Instant)

enum class ClientHealth { NEW, WARMING, READY, JOINED, DISCONNECTED, FAILED, CLOSED }

interface RealtimeClient {
    val id: String
    val health: ClientHealth
    suspend fun initializeMedia()
    suspend fun initializeDataChannel()
    suspend fun setupAndJoin(credential: Credential): VoiceSession
    suspend fun close(reason: String)
}

interface ClientFactory { fun create(): RealtimeClient }
interface CredentialProvider { suspend fun get(key: VoiceKey): Credential }
interface NetworkEpoch { fun current(): Long }
interface Metrics {
    fun span(name: String, attrs: Map<String, String> = emptyMap()): Span
    fun count(name: String, attrs: Map<String, String> = emptyMap())
}
interface Span : AutoCloseable { fun success(); fun fail(t: Throwable) }
data class VoiceSession(val id: String)

data class WarmClient(
    val key: VoiceKey,
    val client: RealtimeClient,
    val createdAt: Instant,
    val networkEpoch: Long
)

class VoiceConnectionManager(
    private val scope: CoroutineScope,
    private val factory: ClientFactory,
    private val credentials: CredentialProvider,
    private val networkEpoch: NetworkEpoch,
    private val metrics: Metrics,
    private val clock: Clock = Clock.systemUTC(),
    private val warmTtl: Duration = Duration.ofMinutes(2)
) {
    private val mutex = Mutex()
    private var cached: WarmClient? = null
    private var inFlight: Deferred<WarmClient>? = null
    private val generation = AtomicLong(0)

    /** Safe to call from multiple product entry points. */
    suspend fun warmup(key: VoiceKey): WarmClient {
        mutex.withLock {
            cached?.takeIf { it.isReusable(key) }?.let {
                metrics.count("voice.warmup.hit")
                return it
            }
            cached?.let { stale -> scope.launch { stale.client.close("cache_invalid") } }
            cached = null

            inFlight?.let {
                metrics.count("voice.warmup.coalesced")
                return it.await()
            }

            val myGeneration = generation.incrementAndGet()
            val deferred = scope.async(start = CoroutineStart.LAZY) {
                createWarmClient(key, myGeneration)
            }
            inFlight = deferred
            deferred.start()
        }

        return try {
            val result = mutex.withLock { inFlight }.let { requireNotNull(it).await() }
            mutex.withLock {
                if (generation.get() == result.client.id.hashCode().toLong()) {
                    // Production code should carry generation explicitly; simplified below.
                }
                cached = result
                inFlight = null
            }
            result
        } catch (t: Throwable) {
            mutex.withLock { inFlight = null }
            metrics.count("voice.warmup.failed", mapOf("type" to t.javaClass.simpleName))
            throw t
        }
    }

    suspend fun join(key: VoiceKey): VoiceSession {
        val overall = metrics.span("voice.click_to_ready")
        try {
            var warm = warmup(key)
            if (!warm.isReusable(key)) { // network/user/config may change while awaiting
                invalidate("changed_before_join")
                warm = warmup(key)
            }
            val credential = withTimeout(1_500) { credentials.get(key) }
            check(credential.expiresAt.isAfter(clock.instant().plusSeconds(30))) {
                "Credential too close to expiry"
            }
            val session = withTimeout(5_000) {
                warm.client.setupAndJoin(credential)
            }
            overall.success()
            return session
        } catch (t: Throwable) {
            overall.fail(t)
            invalidate("join_failed:${t.javaClass.simpleName}")
            throw t
        } finally { overall.close() }
    }

    suspend fun invalidate(reason: String) {
        val victim = mutex.withLock {
            generation.incrementAndGet()
            inFlight?.cancel(CancellationException(reason))
            inFlight = null
            cached.also { cached = null }
        }
        victim?.client?.close(reason)
        metrics.count("voice.cache.invalidated", mapOf("reason" to reason))
    }

    private suspend fun createWarmClient(key: VoiceKey, gen: Long): WarmClient {
        val span = metrics.span("voice.warmup", mapOf("generation" to gen.toString()))
        val client = factory.create()
        try {
            // Independent work runs concurrently; failure cancels siblings.
            coroutineScope {
                val media = async { withTimeout(2_500) { client.initializeMedia() } }
                val data = async { withTimeout(2_500) { client.initializeDataChannel() } }
                awaitAll(media, data)
            }
            check(client.health == ClientHealth.READY)
            span.success()
            return WarmClient(key, client, clock.instant(), networkEpoch.current())
        } catch (t: Throwable) {
            span.fail(t)
            withContext(NonCancellable) { client.close("warmup_failed") }
            throw t
        } finally { span.close() }
    }

    private fun WarmClient.isReusable(requested: VoiceKey): Boolean {
        val age = Duration.between(createdAt, clock.instant())
        return key == requested &&
            age < warmTtl &&
            networkEpoch == this@VoiceConnectionManager.networkEpoch.current() &&
            client.health == ClientHealth.READY
    }
}
代码审查时主动指出：示例中的 generation 结果绑定被刻意简化；生产实现应让 WarmClient 携带 generation，并在写回 cache 时比较，防止旧任务晚到覆盖新状态。这是很好的面试加分点。

进一步优化

  1. 预测式预热：仅在 Voice 入口曝光、用户高概率点击且设备资源允许时预热。
  2. 预算控制：按内存级别、低电量、后台状态决定是否保留媒体 client。
  3. 分层缓存：SDK 初始化可长缓存；token 短缓存；media session 极短 TTL，避免一刀切。
  4. 自适应超时：按网络类型和历史分位数设置，而不是固定魔法数字。
  5. hedged retry：仅对幂等阶段、且 P99 尾延迟严重时谨慎启用，防止连接风暴。
  6. 服务端协同：SetupAndJoin 支持幂等 request id、server drain 和 resume token。

测试清单

  • 10 个并发 warmup 只创建 1 个 client
  • 网络 epoch 改变后旧 client 不可复用
  • 账号/tenant/config 变化使 cache 失效
  • TTL 到期、warmup timeout、join timeout 均正确 close
  • 旧 generation 晚返回不能覆盖新 cache
  • 取消 scope 后无泄漏、无悬挂 deferred
  • credential 距过期不足阈值时刷新

3.1 并发正确性审计

示例最关键的不是 API，而是状态发布的原子性。不要在持有 Mutex 时等待长 IO；典型模式是锁内读取/创建 Deferred，锁外 await，再锁内带 generation 条件提交。否则会阻塞 invalidate/join，甚至死锁。

private data class Flight(val generation: Long, val deferred: Deferred<WarmClient>)

suspend fun warmup(key: VoiceKey): WarmClient {
    val flight = mutex.withLock {
        cached?.takeIf { it.isReusable(key) }?.let { return it }
        inFlight?.let { return@withLock it }
        val g = generation.incrementAndGet()
        Flight(g, scope.async(start = CoroutineStart.LAZY) { createWarmClient(key, g) })
            .also { inFlight = it; it.deferred.start() }
    }
    val result = flight.deferred.await() // no lock held
    return mutex.withLock {
        if (generation.get() == flight.generation && inFlight === flight) {
            cached = result
            inFlight = null
            result
        } else {
            scope.launch { result.client.close("superseded") }
            cached?.takeIf { it.isReusable(key) } ?: throw CancellationException("superseded")
        }
    }
}

3.2 Cancellation 语义

场景期望
一个调用者取消若 warmup 是共享资源，不应必然取消其他调用者的 flight；使用 manager scope
账号登出/manager close取消 in-flight、关闭 cached client，等待短清理
join timeout标记 client health，不能默认继续复用
close 被取消在有界 NonCancellable 中释放 native 资源，禁止无限等待

Structured concurrency 不是“所有 async 都放 GlobalScope”。共享 warmup 可以有独立但有生命周期的 application/session scope；logout 或 host shutdown 必须关闭。

3.3 StateFlow、SharedFlow 与事件

  • 连接状态用 StateFlow：新订阅者需要最新 snapshot。
  • 一次性导航/toast 不应塞进 state 后用 consumed flag；可用 Channel/SharedFlow，但要定义丢失语义。
  • 高频 audio level 不应直接驱动全页面 StateFlow；采样、distinct 和局部订阅。
  • 多个源更新状态时使用 reducer/atomic update，避免 read-modify-write 丢失。

fun `ten callers share one warmup and expired cache rebuilds`() = runTest {

    val factory = ControllableFactory()
    val manager = createManager(backgroundScope, factory = factory)

    manager.invalidate("network_changed")
    val fresh = async { manager.warmup(KEY) }

    assertEquals(factory.freshId, fresh.await().client.id)
    assertFailsWith<CancellationException> { old.await() }
}

3.5 Native 资源与泄漏

WebRTC/媒体 client 常包装 native pointer、线程和 audio device。Kotlin GC 不代表 native 已释放。Manager close 要幂等，按 DataChannel→Peer/Session→AudioDevice→Factory 的依赖顺序释放，并在 debug build 记录未 close 的实例。不要依赖 finalizer。

3.6 拷打

为什么 Mutex 不是端到端幂等？

Mutex 只保护当前进程内临界区；进程重启、多设备、服务端重试都绕过它。它防重复 client 初始化，但邮件发送等业务幂等必须由服务端唯一键和操作状态机保证。

supervisorScope 什么时候用？

如果 media 和 DataChannel 任一失败都使 warmup 无效，应使用 coroutineScope 让兄弟取消。只有各子任务可独立降级、失败不应取消其他任务时才用 supervisorScope，并显式收集每个结果。

为何 Deferred 缓存容易出错？

失败 Deferred 会永久重放异常；取消语义可能被第一个调用者传播；晚完成结果可能覆盖新 generation。因此需要清理失败 flight、独立 scope 和提交时 generation check。

3.7 口述版本

这个 ConnectionManager 的核心是把 warmup 做成带生命周期的 single-flight。锁内只决定复用、加入现有 flight 或创建新 flight，所有网络等待都在锁外；完成后带 generation 再次加锁提交，避免旧任务覆盖网络切换后的新 client。缓存可复用条件包含 user/tenant/config/TTL/network epoch/health。取消分调用者取消和 manager invalidation，native close 在有界 NonCancellable 中完成。测试用虚拟时间和可控 factory 覆盖并发、过期、晚到结果和资源释放。

3.8 代码拷打的证据标准

  • 能在白板上写出锁内/锁外边界，而不是背现成代码。
  • 能指出示例中 generation、共享 Deferred、调用者取消的潜在 bug。
  • 能写一个 deterministic race test，而不是只说“加 Mutex”。
  • 能解释 native close、scope owner 和泄漏检测。
